# Idea Bot Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Telegram-бот, который принимает идеи (текст, голос, форвард), разбирает их через `claude -p` и раскладывает в холст «Творческие идеи» Obsidian с синком в холсты L2/L3.

**Architecture:** Один asyncio-процесс под launchd: aiogram-поллинг → склейка сообщений (Collector) → SQLite → очередь (Pipeline) → whisper → Claude возвращает JSON → Writer детерминированно пишет .md/.canvas напрямую в файлы vault. Claude только думает, Python только пишет.

**Tech Stack:** Python 3.12, uv, aiogram 3.31, PyYAML, python-dotenv, sqlite3, pytest; внешние бинари `claude` (`~/.npm-global/bin/claude`), `mlx_whisper` (`~/.local/bin/mlx_whisper`).

**Spec:** `docs/specs/2026-09-26-idea-bot-design.md`

## Global Constraints

- Проект: `~/Claude Code/idea-bot`, пакет `ideabot`, запуск `python -m ideabot`.
- Vault: `/Users/cucaramba/Documents/Obsidian storage` (env `VAULT_DIR`). Тесты живой vault не трогают никогда, только `tmp_path`.
- Все записи в vault атомарные: `vault/fs.py::atomic_write` (tmp рядом + `os.replace`).
- Canvas сохраняется `json.dumps(..., ensure_ascii=False, indent=1)`, неизвестные поля нод сохраняются как есть.
- Имя файла заметки: `:` → ` —`, `/` → `-` (правило vault).
- Доступ к боту только `OWNER_ID`.
- Claude вызывается только через `brain.run_claude`; при `tools` всегда передаётся и `--allowedTools` с тем же списком.
- В текстах, которые видит Некит (сообщения бота, промпт для заметок), не использовать длинное тире «—» в прозе. Исключение: имена файлов vault по правилу выше и существующие названия.
- Коммит после каждой задачи, сообщение заканчивается строкой `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Тесты: `uv run pytest -q`. Async-код в тестах гоняется через `asyncio.run`, без pytest-asyncio.

## Review Focus

- Заголовок от Claude с символами `: / # ? [ ]` или длиной 200+ символов → валидное уникальное имя файла, заметка открывается (тест в Task 2).
- Холст, который Некит правил руками (text-ноды, `color`, свои поля) → после записи бота все чужие ноды и поля на месте (тест в Task 3).
- Claude вернул несуществующий L2/L3, другой регистр, дубль L2 или тему вне списка → невалидное отбрасывается, явные `#хэштеги` всё равно в связях (тест в Task 6).
- Некит удалил заметку-оригинал между вопросом «дубль?» и нажатием «Дополнить» → создаётся новая заметка, а не вечный write_failed (тест в Task 9).
- Бот упал посреди обработки → после рестарта идея со статусом `new` снова в очереди, а частично записанная `write_failed` откатывается и пишется заново без дублей нод (тест в Task 9).

`ideabot/models.py`:
```python
"""Данные, которые ходят между ботом, очередью, Claude и записью в vault."""
from __future__ import annotations

import json
from dataclasses import asdict, dataclass, field

TYPES = ("бизнес", "контент", "мысль", "прочее")


@dataclass
class Part:
    """Одно сообщение из Telegram, уже без aiogram-объектов."""
    text: str = ""
    origin: str | None = None   # None = написал сам Некит, иначе источник форварда
    photo: str | None = None    # путь к скачанному фото
    voice: str | None = None    # путь к аудио, пока не расшифровано
    was_voice: bool = False
    has_video: bool = False


@dataclass
class Incoming:
    """Идея = склейка нескольких сообщений."""
    parts: list[Part] = field(default_factory=list)

    def to_json(self) -> str:
        return json.dumps([asdict(p) for p in self.parts], ensure_ascii=False)

    @classmethod
    def from_json(cls, s: str) -> "Incoming":
        return cls([Part(**d) for d in json.loads(s)])

    def own_text(self) -> str:
        return "\n\n".join(p.text.strip() for p in self.parts if p.origin is None and p.text.strip())

    def forwards(self) -> list[tuple[str, str, bool]]:
        """[(источник, текст, было ли видео)] с сохранением порядка источников."""
        acc: dict[str, list] = {}
        for p in self.parts:
            if p.origin is None:
                continue
            entry = acc.setdefault(p.origin, [[], False])
            if p.text.strip():
                entry[0].append(p.text.strip())
            entry[1] = entry[1] or p.has_video
        return [(o, "\n\n".join(texts), video) for o, (texts, video) in acc.items()]

    def photos(self) -> list[str]:
        return [p.photo for p in self.parts if p.photo]

    def source(self) -> str:
        if any(p.origin is not None for p in self.parts):
            return "форвард"
        if any(p.was_voice for p in self.parts):
            return "голос"
        return "текст"

    def is_empty(self) -> bool:
        return not (self.own_text() or self.forwards() or self.photos() or any(p.voice for p in self.parts))


@dataclass
class Link:
    l2: str
    l3: str | None = None
    l3_kind: str = "none"  # file | group | none


@dataclass
class Analysis:
    type: str
    theme: str
    title: str
    body_md: str
    score: int | None = None
    links: list[Link] = field(default_factory=list)
    duplicate_of: str | None = None
    unknown_tags: list[str] = field(default_factory=list)

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: dict) -> "Analysis":
        d = dict(d)
        d["links"] = [Link(**x) for x in d.get("links") or []]
        return cls(**d)
```

```python
"""Атомарная запись и wikilinks."""
from __future__ import annotations

import os
import tempfile
from pathlib import Path


def atomic_write(path: Path, text: str) -> None:
    """Пишем во временный файл рядом и подменяем: Obsidian никогда не увидит полфайла."""
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=path.parent, prefix=".ideabot-", suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            f.write(text)
        os.replace(tmp, path)
    except BaseException:
        Path(tmp).unlink(missing_ok=True)
        raise


def wikilink(rel: str) -> str:
    """Путь от корня vault -> [[путь|имя]]. Путь нужен, потому что имена бывают не уникальны."""
    target = rel[:-3] if rel.endswith(".md") else rel
    name = target.rsplit("/", 1)[-1]
    return f"[[{target}|{name}]]" if "/" in target else f"[[{target}]]"
```

```python
"""Заметка идеи: имя файла, рендер, секции, MOC."""
from __future__ import annotations

import json
import re
from pathlib import Path

from ..models import Incoming
from .fs import atomic_write

LINKS_H = "## Связанные заметки"
_BAD = re.compile(r'[\\*?"<>|#^\[\]]')


def safe_filename(title: str) -> str:
    name = title.replace(":", " —").replace("/", "-")
    name = _BAD.sub("", name)
    name = re.sub(r"\s+", " ", name).strip().strip(".").strip()
    return name[:100].rstrip() or "Идея"


def unique_note_path(folder: Path, title: str) -> Path:
    base = safe_filename(title)
    path, n = folder / f"{base}.md", 2
    while path.exists():
        path, n = folder / f"{base} ({n}).md", n + 1
    return path


def quote(text: str) -> str:
    return "\n".join(f"> {line}" if line else ">" for line in text.splitlines())


def render_source(inc: Incoming) -> str:
    blocks = []
    if own := inc.own_text():
        blocks.append(own)
    for origin, text, video in inc.forwards():
        body = text or "(без текста)"
        if video:
            body += "\n(видео не смотрел, только подпись)"
        blocks.append(f"Форвард от «{origin}»:\n{body}")
    if photos := inc.photos():
        blocks.append(f"(фото: {len(photos)} шт.)")
    return "\n\n".join(blocks) or "(пусто)"


def _yaml(v) -> str:
    return str(v) if isinstance(v, (int, float)) else json.dumps(str(v), ensure_ascii=False)


def _links_block(links: list[str]) -> str:
    return "\n".join(f"- {x}" for x in links)


def render_note(meta: dict, title: str, body_md: str, source_text: str, links: list[str]) -> str:
    fm = "---\n" + "".join(f"{k}: {_yaml(v)}\n" for k, v in meta.items() if v is not None) + "---\n"
    return (f"{fm}# {title}\n\n{body_md.strip()}\n\n> [!quote]- Исходник\n{quote(source_text)}\n\n"
            f"{LINKS_H}\n{_links_block(links)}\n")


def get_links(text: str) -> list[str]:
    i = text.find(LINKS_H)
    if i == -1:
        return []
    out = []
    for line in text[i + len(LINKS_H):].splitlines():
        if line.startswith("## "):
            break
        if line.startswith("- "):
            out.append(line[2:].strip())
    return out


def set_links(text: str, links: list[str]) -> str:
    i = text.find(LINKS_H)
    if i == -1:
        return text.rstrip("\n") + f"\n\n{LINKS_H}\n{_links_block(links)}\n"
    j = text.find("\n## ", i + len(LINKS_H))
    tail = text[j:] if j != -1 else ""
    return text[:i] + f"{LINKS_H}\n{_links_block(links)}\n" + tail


def add_section(text: str, heading: str, body: str) -> str:
    """Секция встаёт перед «Связанными заметками», чтобы они всегда были в конце."""
    block = f"## {heading}\n\n{body.strip()}\n\n"
    i = text.find(LINKS_H)
    if i == -1:
        return text.rstrip("\n") + "\n\n" + block
    return text[:i] + block + text[i:]


def remove_section(text: str, heading: str) -> str:
    h = f"## {heading}\n"
    i = text.find(h)
    if i == -1:
        return text
    j = text.find("\n## ", i + len(h))
    return text[:i] + (text[j + 1:] if j != -1 else "")


def moc_add(moc: Path, theme: str, item: str) -> None:
    text = moc.read_text("utf-8") if moc.exists() else f"# {moc.stem}\n"
    lines = text.splitlines()
    head = f"## {theme}"
    hi = next((i for i, line in enumerate(lines) if line.strip() == head), None)
    if hi is None:
        atomic_write(moc, text.rstrip("\n") + f"\n\n{head}\n\n- {item}\n")
        return
    end = hi + 1
    while end < len(lines) and not lines[end].startswith("## "):
        end += 1
    last = end
    while last > hi + 1 and not lines[last - 1].strip():
        last -= 1
    lines.insert(last, f"- {item}")
    atomic_write(moc, "\n".join(lines) + "\n")


def moc_remove(moc: Path, item: str) -> None:
    """Убирает пункт. Если в его разделе больше ничего нет, убирает и заголовок раздела."""
    if not moc.exists():
        return
    out = []
    for sec in re.split(r"(?m)^(?=## )", moc.read_text("utf-8")):
        lines = sec.splitlines(keepends=True)
        kept = [x for x in lines if x.strip() != f"- {item}"]
        if len(kept) != len(lines) and sec.startswith("## ") and not any(x.strip() for x in kept[1:]):
            continue
        out.append("".join(kept))
    atomic_write(moc, "".join(out).rstrip("\n") + "\n")
```

```python
"""Холст Obsidian (JSON Canvas) и раскладка групп и нод."""
from __future__ import annotations

import json
import uuid
from pathlib import Path

from .fs import atomic_write

NOTE_W, NOTE_H, GAP = 400, 120, 20
PAD_X, PAD_TOP, PAD_BOTTOM = 30, 50, 30
GROUP_W, COL_STEP, ROW_GAP, SIDE_GAP = 460, 500, 120, 200
IDEAS_GROUP = "💡 Идеи"


def new_id() -> str:
    return uuid.uuid4().hex[:16]


def _inside(n: dict, g: dict) -> bool:
    return (n["x"] >= g["x"] and n["y"] >= g["y"]
            and n["x"] + n["width"] <= g["x"] + g["width"] and n["y"] + n["height"] <= g["y"] + g["height"])


class Canvas:
    def __init__(self, path: Path, data: dict):
        self.path = path
        self.data = data
        data.setdefault("nodes", [])
        data.setdefault("edges", [])

    @classmethod
    def load(cls, path: Path) -> "Canvas":
        data = json.loads(path.read_text("utf-8")) if path.exists() else {"nodes": [], "edges": []}
        return cls(path, data)

    def save(self) -> None:
        atomic_write(self.path, json.dumps(self.data, ensure_ascii=False, indent=1))

    @property
    def nodes(self) -> list[dict]:
        return self.data["nodes"]

    @property
    def edges(self) -> list[dict]:
        return self.data["edges"]

    def node(self, nid: str) -> dict | None:
        return next((n for n in self.nodes if n.get("id") == nid), None)

    def groups(self) -> list[dict]:
        return [n for n in self.nodes if n.get("type") == "group"]

    def group(self, label: str) -> dict | None:
        return next((g for g in self.groups() if g.get("label") == label), None)

    def file_node(self, rel: str) -> dict | None:
        return next((n for n in self.nodes if n.get("type") == "file" and n.get("file") == rel), None)

    def children(self, g: dict) -> list[dict]:
        return [n for n in self.nodes if n is not g and n.get("type") != "group" and _inside(n, g)]

    def bounds(self, skip=frozenset()) -> tuple[int, int, int, int] | None:
        ns = [n for n in self.nodes if n.get("id") not in skip]
        if not ns:
            return None
        return (min(n["x"] for n in ns), min(n["y"] for n in ns),
                max(n["x"] + n["width"] for n in ns), max(n["y"] + n["height"] for n in ns))

    def add_node(self, **fields) -> dict:
        n = {"id": new_id(), **fields}
        self.nodes.append(n)
        return n

    def add_edge(self, from_id: str, to_id: str) -> dict:
        e = {"id": new_id(), "fromNode": from_id, "toNode": to_id}
        self.edges.append(e)
        return e

    def remove_node(self, nid: str) -> None:
        self.data["nodes"] = [n for n in self.nodes if n.get("id") != nid]
        self.data["edges"] = [e for e in self.edges if nid not in (e.get("fromNode"), e.get("toNode"))]

    def remove_edge(self, eid: str) -> None:
        self.data["edges"] = [e for e in self.edges if e.get("id") != eid]

    def move(self, g: dict, x: int, y: int) -> None:
        dx, dy = x - g["x"], y - g["y"]
        for n in [g, *self.children(g)]:
            n["x"] += dx
            n["y"] += dy


def _theme_row_y(cv: Canvas, themes: list[str]) -> int:
    skip: set[str] = set()
    for g in cv.groups():
        if g.get("label") in themes:
            skip.add(g["id"])
            skip.update(c["id"] for c in cv.children(g))
    b = cv.bounds(skip)
    return 0 if b is None else b[3] + ROW_GAP


def ensure_theme_group(cv: Canvas, label: str, themes: list[str]) -> dict:
    """Группа темы в ряду под всем остальным. Старую группу с тем же именем переносим в ряд."""
    row_y = _theme_row_y(cv, themes)
    x = themes.index(label) * COL_STEP
    g = cv.group(label)
    if g is None:
        return cv.add_node(type="group", label=label, x=x, y=row_y, width=GROUP_W, height=PAD_TOP + PAD_BOTTOM)
    if g["y"] < row_y:
        cv.move(g, x, row_y)
    return g


def ensure_side_group(cv: Canvas, label: str = IDEAS_GROUP) -> dict:
    """Группа «💡 Идеи» справа от всего содержимого L2-холста."""
    if g := cv.group(label):
        return g
    b = cv.bounds()
    x, y = (0, 0) if b is None else (b[2] + SIDE_GAP, b[1])
    return cv.add_node(type="group", label=label, x=x, y=y, width=GROUP_W, height=PAD_TOP + PAD_BOTTOM)


def add_to_group(cv: Canvas, g: dict, rel: str) -> dict:
    kids = cv.children(g)
    y = max((k["y"] + k["height"] for k in kids), default=g["y"] + PAD_TOP - GAP) + GAP
    n = cv.add_node(type="file", file=rel, x=g["x"] + PAD_X, y=y, width=NOTE_W, height=NOTE_H)
    g["height"] = max(g["height"], y + NOTE_H + PAD_BOTTOM - g["y"])
    return n


def drop_empty_side_group(cv: Canvas) -> None:
    g = cv.group(IDEAS_GROUP)
    if g and not cv.children(g):
        cv.remove_node(g["id"])
```

```python
"""Что Claude знает о vault: холсты L2, их группы и заметки L3, уже записанные идеи, цели."""
from __future__ import annotations

import json
import logging
from dataclasses import dataclass
from pathlib import Path

from ..config import Settings
from .canvas import IDEAS_GROUP

log = logging.getLogger(__name__)


@dataclass
class L2Info:
    name: str
    canvas: Path
    moc: str | None
    notes: dict[str, tuple[str, str]]
    groups: dict[str, str]
    shallow: bool


@dataclass
class VaultIndex:
    l2: dict[str, L2Info]
    ideas: list[str]
    goals: str

    def l2_by_tag(self, tag: str) -> str | None:
        norm = tag.replace("_", " ").casefold()
        return next((name for name in self.l2 if name.casefold() == norm), None)


def build_index(s: Settings) -> VaultIndex:
    v = s.vault
    l2: dict[str, L2Info] = {}
    for p in sorted(v.glob("*.canvas")):
        if p.stem == s.ideas_canvas:
            continue
        try:
            data = json.loads(p.read_text("utf-8"))
        except (OSError, ValueError) as e:
            log.warning("не прочитал %s: %s", p.name, e)
            continue
        moc_name = f"{p.stem} — карта заметок"
        moc, notes, groups = None, {}, {}
        for n in data.get("nodes", []):
            if n.get("type") == "group" and n.get("label") and n["label"] != IDEAS_GROUP:
                groups[n["label"]] = n["id"]
            elif n.get("type") == "file" and str(n.get("file", "")).endswith(".md"):
                rel = n["file"]
                stem = Path(rel).stem
                if stem == moc_name:
                    moc = rel
                elif not rel.startswith(s.ideas_canvas + "/"):  # ноды-синки идей это не L3
                    notes.setdefault(stem, (n["id"], rel))
        if moc is None:
            moc = next((c for c in (f"{p.stem}/{moc_name}.md", f"{moc_name}.md") if (v / c).exists()), None)
        shallow = p.stem in s.shallow_canvases
        l2[p.stem] = L2Info(p.stem, p, moc, {} if shallow else notes, groups, shallow)
    ideas = []
    if s.ideas_dir.exists():
        ideas = sorted(f.stem for f in s.ideas_dir.glob("*.md") if f.name != s.moc_path.name)
    goals_p = v / s.goals_note
    goals = goals_p.read_text("utf-8") if goals_p.exists() else ""
    return VaultIndex(l2, ideas, goals)


def _render(index: VaultIndex, drop: set[str], only: set[str] | None) -> str:
    lines = ["## Холсты L2"]
    for info in index.l2.values():
        if only is not None and info.name not in only:
            continue
        lines.append(f"### {info.name}" + (" (только группы)" if info.shallow or info.name in drop else ""))
        if info.groups:
            lines.append("Группы: " + "; ".join(info.groups))
        if info.notes and info.name not in drop:
            lines.append("Заметки: " + "; ".join(info.notes))
    return "\n".join(lines)


def render_for_prompt(index: VaultIndex, max_chars: int, only: set[str] | None = None) -> str:
    """Не влезает в лимит: у самых больших холстов оставляем только группы."""
    drop: set[str] = set()
    text = _render(index, drop, only)
    for info in sorted(index.l2.values(), key=lambda i: len(i.notes), reverse=True):
        if len(text) <= max_chars or not info.notes:
            break
        drop.add(info.name)
        text = _render(index, drop, only)
    return text
```

```python
"""SQLite: какое сообщение стало какой заметкой и какими нодами. Нужно для отмены, перевеса, повторов."""
from __future__ import annotations

import json
import sqlite3
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path

from .models import Analysis, Incoming

SCHEMA = """
CREATE TABLE IF NOT EXISTS ideas(
  id INTEGER PRIMARY KEY,
  created_at TEXT NOT NULL,
  status TEXT NOT NULL,
  chat_id INTEGER NOT NULL,
  card_msg_id INTEGER,
  payload TEXT NOT NULL,
  analysis TEXT,
  note_path TEXT,
  mode TEXT,
  append_to TEXT,
  append_heading TEXT,
  deep INTEGER NOT NULL DEFAULT 0,
  error TEXT
);
CREATE TABLE IF NOT EXISTS links(
  id INTEGER PRIMARY KEY,
  idea_id INTEGER NOT NULL,
  canvas TEXT,
  node_id TEXT,
  node_created INTEGER NOT NULL DEFAULT 0,
  edge_id TEXT,
  wikilink TEXT
);
"""
UPDATABLE = {"status", "card_msg_id", "payload", "analysis", "note_path", "mode",
             "append_to", "append_heading", "deep", "error"}


@dataclass
class Idea:
    id: int
    created_at: str
    status: str
    chat_id: int
    card_msg_id: int | None
    payload: str
    analysis: str | None
    note_path: str | None
    mode: str | None
    append_to: str | None
    append_heading: str | None
    deep: int
    error: str | None

    @property
    def incoming(self) -> Incoming:
        return Incoming.from_json(self.payload)

    @property
    def parsed(self) -> Analysis | None:
        return Analysis.from_dict(json.loads(self.analysis)) if self.analysis else None


class DB:
    def __init__(self, path: Path):
        path.parent.mkdir(parents=True, exist_ok=True)
        self.c = sqlite3.connect(path)
        self.c.row_factory = sqlite3.Row
        self.c.executescript(SCHEMA)

    def create(self, chat_id: int, inc: Incoming) -> int:
        cur = self.c.execute(
            "INSERT INTO ideas(created_at, status, chat_id, payload) VALUES(?, 'new', ?, ?)",
            (datetime.now().isoformat(timespec="seconds"), chat_id, inc.to_json()))
        self.c.commit()
        return cur.lastrowid

    def get(self, idea_id: int) -> Idea | None:
        row = self.c.execute("SELECT * FROM ideas WHERE id=?", (idea_id,)).fetchone()
        return Idea(**dict(row)) if row else None

    def update(self, idea_id: int, **fields) -> None:
        if bad := set(fields) - UPDATABLE:
            raise ValueError(f"нельзя обновлять {bad}")
        sets = ", ".join(f"{k}=?" for k in fields)
        self.c.execute(f"UPDATE ideas SET {sets} WHERE id=?", (*fields.values(), idea_id))
        self.c.commit()

    def by_status(self, *statuses: str) -> list[Idea]:
        q = ",".join("?" * len(statuses))
        rows = self.c.execute(f"SELECT * FROM ideas WHERE status IN ({q}) ORDER BY id", statuses).fetchall()
        return [Idea(**dict(r)) for r in rows]

    def add_link(self, idea_id: int, canvas: str | None, node_id: str | None = None,
                 node_created: bool = False, edge_id: str | None = None, wikilink: str | None = None) -> None:
        self.c.execute(
            "INSERT INTO links(idea_id, canvas, node_id, node_created, edge_id, wikilink) VALUES(?,?,?,?,?,?)",
            (idea_id, canvas, node_id, int(node_created), edge_id, wikilink))
        self.c.commit()

    def links(self, idea_id: int) -> list[dict]:
        rows = self.c.execute("SELECT * FROM links WHERE idea_id=? ORDER BY id", (idea_id,)).fetchall()
        return [dict(r) for r in rows]

    def delete_links(self, ids: list[int]) -> None:
        self.c.executemany("DELETE FROM links WHERE id=?", [(i,) for i in ids])
        self.c.commit()
```

```python
async def run_claude(s: Settings, prompt: str, *, system: str, schema: dict | None = None,
                     tools: tuple[str, ...] = (), add_dirs: tuple[str, ...] = (), timeout: int) -> dict | str:
    args = [s.claude_bin, "-p", "--output-format", "json", "--no-session-persistence",
            "--system-prompt", system, "--tools", ",".join(tools)]
    if tools:
        args += ["--allowedTools", ",".join(tools)]  # без этого в -p тул упрётся в запрос разрешения
    for d in add_dirs:
        args += ["--add-dir", d]
    if schema is not None:
        args += ["--json-schema", json.dumps(schema, ensure_ascii=False)]
    s.data_dir.mkdir(parents=True, exist_ok=True)
    env = {**os.environ, "PATH": f"{os.path.dirname(s.claude_bin)}:/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin"}
    proc = await asyncio.create_subprocess_exec(
        *args, stdin=asyncio.subprocess.PIPE, stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE,
        cwd=str(s.data_dir), env=env)
    try:
        out, err = await asyncio.wait_for(proc.communicate(prompt.encode()), timeout)
    except asyncio.TimeoutError:
        proc.kill()
        await proc.wait()
        raise BrainError(f"claude не ответил за {timeout} с")
    if proc.returncode != 0:
        raise BrainError(f"claude exit {proc.returncode}: {err.decode(errors='ignore')[-300:]}")
    try:
        data = json.loads(out)
    except ValueError:
        raise BrainError("claude вернул не JSON")
    if data.get("is_error"):
        raise BrainError(f"claude: {str(data.get('result'))[:300]}")
    if schema is not None:
        so = data.get("structured_output")
        if not isinstance(so, dict):
            raise BrainError("claude не вернул structured_output")
        return so
    return str(data.get("result") or "")


def explicit_l2(inc: Incoming, index: VaultIndex) -> tuple[list[str], list[str]]:
    found, unknown = [], []
    for tag in re.findall(r"#([\w-]+)", inc.own_text()):
        name = index.l2_by_tag(tag)
        if name and name not in found:
            found.append(name)
        elif not name and tag not in unknown:
            unknown.append(tag)
    return found, unknown


def clean_links(items: list[dict], index: VaultIndex) -> list[Link]:
    out, seen = [], set()
    for it in items:
        l2 = it.get("l2")
        info = index.l2.get(l2)
        if info is None or l2 in seen:
            continue
        seen.add(l2)
        l3 = it.get("l3")
        if l3 in info.notes:
            out.append(Link(l2, l3, "file"))
        elif l3 in info.groups:
            out.append(Link(l2, l3, "group"))
        else:
            out.append(Link(l2))
    return out


def demote(md: str) -> str:
    """Заголовки # и ## из ответа Claude -> ###: иначе они сломают секции заметки."""
    return re.sub(r"(?m)^#{1,2} ", "### ", md)


def validate(raw: dict, index: VaultIndex, s: Settings, explicit: list[str], unknown: list[str]) -> Analysis:
    t = raw.get("type") if raw.get("type") in TYPES else "прочее"
    theme = raw.get("theme") if raw.get("theme") in s.themes else s.fallback_theme
    title = (raw.get("title") or "").strip()[:80].strip() or "Идея без названия"
    score = raw.get("score")
    score = score if t == "бизнес" and isinstance(score, int) and 1 <= score <= 10 else None
    links = clean_links(raw.get("links") or [], index)
    for name in explicit:
        if name not in {l.l2 for l in links}:
            links.append(Link(name))
    dup = raw.get("duplicate_of")
    return Analysis(type=t, theme=theme, title=title, body_md=demote((raw.get("body_md") or "").strip()), score=score,
                    links=links, duplicate_of=dup if dup in index.ideas else None, unknown_tags=unknown)


class Brain:
    def __init__(self, s: Settings):
        self.s = s

    async def analyze(self, inc: Incoming, index: VaultIndex) -> Analysis:
        explicit, unknown = explicit_l2(inc, index)
        photos = inc.photos()
        parts = [
            "Темы: " + ", ".join(self.s.themes),
            "Явно указанные холсты: " + (", ".join(explicit) or "нет"),
            "", render_for_prompt(index, self.s.max_index_chars),
            "", "## Уже существующие идеи", "\n".join(index.ideas) or "(нет)",
        ]
        if index.goals:
            parts += ["", "## Цели Некита", index.goals[:4000]]
        parts += ["", "## Идея", render_source(inc)]
        if photos:
            parts += ["", "Фото к идее, посмотри их тулом Read:", *photos]
        raw = await run_claude(
            self.s, "\n".join(parts), system=SYSTEM, schema=ANALYSIS_SCHEMA,
            tools=("Read",) if photos else (), add_dirs=tuple(sorted({str(Path(p).parent) for p in photos})),
            timeout=self.s.claude_timeout)
        return validate(raw, index, self.s, explicit, unknown)

    async def pick_l3(self, a: Analysis, l2_names: list[str], index: VaultIndex) -> list[Link]:
        prompt = "\n".join([f"# {a.title}", "", a.body_md, "",
                            render_for_prompt(index, self.s.max_index_chars, only=set(l2_names))])
        raw = await run_claude(self.s, prompt, system=PICK_SYSTEM, schema=LINKS_SCHEMA,
                               timeout=self.s.claude_timeout)
        got = {l.l2: l for l in clean_links(raw.get("links") or [], index)}
        return [got.get(n, Link(n)) for n in l2_names if n in index.l2]

    async def deep_dive(self, a: Analysis) -> str:
        md = await run_claude(self.s, f"# {a.title}\n\n{a.body_md}", system=DEEP_SYSTEM,
                              tools=("WebSearch", "WebFetch"), timeout=self.s.deep_timeout)
        if not md.strip():
            raise BrainError("claude вернул пустой ответ")
        return demote(md.strip())
```

```python
"""Writer: всё, что меняет vault. Каждое изменение холста записывается в БД, чтобы его можно было откатить."""
from __future__ import annotations

from collections import defaultdict
from datetime import date, datetime
from pathlib import Path

from ..config import Settings
from ..db import DB
from ..models import Analysis, Incoming, Link
from .canvas import Canvas, add_to_group, drop_empty_side_group, ensure_side_group, ensure_theme_group
from .fs import atomic_write, wikilink
from .index import VaultIndex
from .notes import (add_section, get_links, moc_add, moc_remove, quote, remove_section, render_note,
                    render_source, set_links, unique_note_path)

RAW_BODY = "#неразобрано\n\nClaude не смог разобрать идею, исходник ниже. Нажми «🔄 Разобрать ещё раз» в боте."


class Writer:
    def __init__(self, s: Settings, db: DB):
        self.s = s
        self.db = db

    def _rel(self, p: Path) -> str:
        return p.relative_to(self.s.vault).as_posix()

    def _wikilinks(self, links: list[Link], index: VaultIndex | None) -> list[str]:
        out: list[str] = []
        for link in links:
            info = index.l2[link.l2]
            if link.l3_kind == "file":
                target = info.notes[link.l3][1]
            else:
                target = info.moc or info.canvas.name
            if (w := wikilink(target)) not in out:
                out.append(w)
        return out

    def save_new(self, idea_id: int, a: Analysis, inc: Incoming, index: VaultIndex | None) -> str:
        path = unique_note_path(self.s.ideas_dir, a.title)
        rel = self._rel(path)
        meta = {"type": a.type, "theme": a.theme, "created": date.today().isoformat(), "source": inc.source(),
                "forward_from": ", ".join(o for o, _, _ in inc.forwards()) or None, "score": a.score,
                "idea_bot_id": idea_id}
        atomic_write(path, render_note(meta, a.title, a.body_md, render_source(inc), self._wikilinks(a.links, index)))
        self.db.update(idea_id, note_path=rel)  # сразу: если дальше упадём, undo найдёт что откатить
        moc_add(self.s.moc_path, a.theme, f"[[{path.stem}]]")
        cv = Canvas.load(self.s.ideas_canvas_path)
        node = add_to_group(cv, ensure_theme_group(cv, a.theme, self.s.themes), rel)
        cv.save()
        self.db.add_link(idea_id, self._rel(cv.path), node_id=node["id"], node_created=True)
        self._sync_l2(idea_id, rel, a.links, index)
        return rel

    def save_raw(self, idea_id: int, inc: Incoming) -> str:
        text = inc.own_text() or next((t for _, t, _ in inc.forwards() if t), "")
        words = " ".join(text.split()[:8])
        a = Analysis(type="прочее", theme=self.s.fallback_theme,
                     title=f"Неразобрано: {words or f'идея {idea_id}'}", body_md=RAW_BODY)
        return self.save_new(idea_id, a, inc, None)

    def _sync_l2(self, idea_id: int, note_rel: str, links: list[Link], index: VaultIndex | None) -> None:
        for link in links:
            info = index.l2[link.l2]
            cv = Canvas.load(info.canvas)
            node = cv.file_node(note_rel)
            created = node is None
            if created:
                node = add_to_group(cv, ensure_side_group(cv), note_rel)
            target = None
            if link.l3_kind == "file" and link.l3 in info.notes:
                target = info.notes[link.l3][0]
            elif link.l3_kind == "group":
                target = info.groups.get(link.l3)
            edge_id = cv.add_edge(node["id"], target)["id"] if target and cv.node(target) else None
            cv.save()
            self.db.add_link(idea_id, self._rel(info.canvas), node_id=node["id"], node_created=created, edge_id=edge_id)

    def _remove_canvas_links(self, rows: list[dict]) -> None:
        by_canvas: dict[str, list[dict]] = defaultdict(list)
        for r in rows:
            if r["canvas"]:
                by_canvas[r["canvas"]].append(r)
        for crel, rs in by_canvas.items():
            path = self.s.vault / crel
            if not path.exists():
                continue
            cv = Canvas.load(path)
            for r in rs:
                if r["edge_id"]:
                    cv.remove_edge(r["edge_id"])
                if r["node_created"] and r["node_id"]:
                    cv.remove_node(r["node_id"])
            drop_empty_side_group(cv)
            cv.save()

    def relink(self, idea_id: int, links: list[Link], index: VaultIndex) -> None:
        idea = self.db.get(idea_id)
        ideas_canvas = self._rel(self.s.ideas_canvas_path)
        rows = [r for r in self.db.links(idea_id) if r["canvas"] and r["canvas"] != ideas_canvas]
        self._remove_canvas_links(rows)
        self.db.delete_links([r["id"] for r in rows])
        path = self.s.vault / idea.note_path
        atomic_write(path, set_links(path.read_text("utf-8"), self._wikilinks(links, index)))
        self._sync_l2(idea_id, idea.note_path, links, index)

    def append(self, idea_id: int, target_title: str, a: Analysis, inc: Incoming, index: VaultIndex) -> str:
        path = self.s.ideas_dir / f"{target_title}.md"
        rel = self._rel(path)
        heading = f"Дополнение ({datetime.now():%Y-%m-%d %H:%M})"
        body = f"{a.body_md.strip()}\n\n> [!quote]- Исходник\n{quote(render_source(inc))}"
        text = add_section(path.read_text("utf-8"), heading, body)
        old = get_links(text)
        new = [x for x in self._wikilinks(a.links, index) if x not in old]
        atomic_write(path, set_links(text, old + new))
        self.db.update(idea_id, note_path=rel, append_to=rel, append_heading=heading)
        for x in new:
            self.db.add_link(idea_id, None, wikilink=x)
        self._sync_l2(idea_id, rel, a.links, index)
        return rel

    def undo(self, idea_id: int) -> None:
        idea = self.db.get(idea_id)
        rows = self.db.links(idea_id)
        self._remove_canvas_links(rows)
        if idea.append_to:
            path = self.s.vault / idea.append_to
            if path.exists():
                text = remove_section(path.read_text("utf-8"), idea.append_heading)
                added = {r["wikilink"] for r in rows if r["wikilink"]}
                atomic_write(path, set_links(text, [x for x in get_links(text) if x not in added]))
        elif idea.note_path:
            path = self.s.vault / idea.note_path
            path.unlink(missing_ok=True)
            moc_remove(self.s.moc_path, f"[[{path.stem}]]")
        self.db.delete_links([r["id"] for r in rows])
        self.db.update(idea_id, note_path=None, append_to=None, append_heading=None)

    def add_deep(self, idea_id: int, md: str) -> None:
        idea = self.db.get(idea_id)
        path = self.s.vault / idea.note_path
        atomic_write(path, add_section(path.read_text("utf-8"), f"Копнули глубже ({date.today().isoformat()})", md))
        self.db.update(idea_id, deep=1)
```

```python
"""Склейка: всё, что пришло подряд с паузами меньше окна, это одна идея (коммент + форвард, альбом)."""
from __future__ import annotations

import asyncio
import logging
from typing import Awaitable, Callable

from .models import Part

log = logging.getLogger(__name__)


class Collector:
    def __init__(self, window: float, on_flush: Callable[[int, list[Part]], Awaitable[None]]):
        self.window = window
        self.on_flush = on_flush
        self._buf: dict[int, list[Part]] = {}
        self._timers: dict[int, asyncio.Task] = {}

    def add(self, chat_id: int, part: Part) -> None:
        self._buf.setdefault(chat_id, []).append(part)
        if t := self._timers.pop(chat_id, None):
            t.cancel()
        self._timers[chat_id] = asyncio.create_task(self._later(chat_id))

    async def _later(self, chat_id: int) -> None:
        await asyncio.sleep(self.window)
        self._timers.pop(chat_id, None)
        parts = self._buf.pop(chat_id, [])
        if parts:
            try:
                await self.on_flush(chat_id, parts)
            except Exception:
                log.exception("не смог принять идею")
```

```python
from __future__ import annotations

import asyncio
import logging
import os
import tempfile
from pathlib import Path

from .config import Settings

log = logging.getLogger(__name__)
_lock = asyncio.Lock()  # две модели параллельно на 8 ГБ не запускаем

PROMPT = "Идея для бизнеса, контента или заметка. Рилс, пост, медиабаинг, личный бренд, Obsidian."


async def transcribe(s: Settings, audio: Path) -> str:
    async with _lock:
        with tempfile.TemporaryDirectory() as out:
            proc = await asyncio.create_subprocess_exec(
                s.whisper_bin, str(audio), "--model", s.whisper_model, "--language", "ru",
                "--output-format", "txt", "--output-dir", out, "--verbose", "False",
                "--condition-on-previous-text", "False", "--initial-prompt", PROMPT,
                stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE,
                env={**os.environ, "PATH": "/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin:" + os.environ.get("PATH", "")},
            )
            try:
                _, err = await asyncio.wait_for(proc.communicate(), timeout=300)
            except asyncio.TimeoutError:
                proc.kill()
                await proc.wait()
                raise RuntimeError("Whisper не успел за 5 минут")
            if proc.returncode != 0:
                log.error("whisper failed: %s", err.decode(errors="ignore")[-500:])
                raise RuntimeError("Whisper не смог расшифровать голос")
            txt = next(Path(out).glob("*.txt"), None)
            return txt.read_text(encoding="utf-8").strip() if txt else ""
```

```python
"""Очередь идей. Главное правило: идея не теряется, при любом сбое она в vault или в БД с кнопкой повтора."""
from __future__ import annotations

import asyncio
import json
import logging
from pathlib import Path
from typing import Awaitable, Callable, Protocol

from .brain import BrainError
from .config import Settings
from .db import DB
from .models import Link
from .vault.index import build_index
from .vault.sync import Writer

log = logging.getLogger(__name__)


class Notifier(Protocol):
    async def refresh(self, idea_id: int) -> None: ...
    async def busy(self, idea_id: int, text: str) -> None: ...
    async def error(self, idea_id: int, text: str) -> None: ...


class Pipeline:
    def __init__(self, s: Settings, db: DB, writer: Writer, brain, stt: Callable[[Path], Awaitable[str]],
                 notifier: Notifier, index_fn=build_index):
        self.s, self.db, self.writer, self.brain, self.stt = s, db, writer, brain, stt
        self.notifier = notifier
        self.index_fn = index_fn
        self.q: asyncio.Queue = asyncio.Queue()
        self.vault_lock = asyncio.Lock()
        self._tasks: set[asyncio.Task] = set()

    def submit(self, idea_id: int) -> None:
        self.q.put_nowait(("process", idea_id, ()))

    def spawn(self, coro) -> None:
        t = asyncio.create_task(coro)
        self._tasks.add(t)
        t.add_done_callback(self._tasks.discard)

    def startup(self) -> None:
        for idea in self.db.by_status("new"):
            self.submit(idea.id)
        for idea in self.db.by_status("write_failed"):
            self.q.put_nowait(("write", idea.id, ()))

    async def run(self) -> None:
        while True:
            kind, idea_id, args = await self.q.get()
            try:
                await getattr(self, kind)(idea_id, *args)
            except Exception:
                log.exception("%s #%s упал", kind, idea_id)
            finally:
                self.q.task_done()

    async def retry_loop(self) -> None:
        while True:
            await asyncio.sleep(self.s.retry_every)
            for idea in self.db.by_status("write_failed"):
                self.q.put_nowait(("write", idea.id, ()))

    async def process(self, idea_id: int) -> None:
        idea = self.db.get(idea_id)
        if idea is None or idea.status not in ("new", "stt_failed", "raw"):
            return
        inc = idea.incoming
        for part in inc.parts:
            if not part.voice:
                continue
            try:
                part.text = ((part.text + "\n\n") if part.text else "") + (await self.stt(Path(part.voice))).strip()
            except Exception as e:
                log.warning("stt #%s: %s", idea_id, e)
                self.db.update(idea_id, status="stt_failed", error=str(e))
                await self.notifier.refresh(idea_id)
                return
            part.voice, part.was_voice = None, True
            self.db.update(idea_id, payload=inc.to_json())
        if inc.is_empty():
            self.db.update(idea_id, status="undone", error="пустое сообщение")
            await self.notifier.refresh(idea_id)
            return
        try:
            a = await self.brain.analyze(inc, self.index_fn(self.s))
        except BrainError as e:
            log.warning("brain #%s: %s", idea_id, e)
            if idea.status != "raw":
                async with self.vault_lock:
                    self.writer.save_raw(idea_id, inc)
            self.db.update(idea_id, status="raw", error=str(e))
            await self.notifier.refresh(idea_id)
            return
        self.db.update(idea_id, analysis=json.dumps(a.to_dict(), ensure_ascii=False), error=None)
        if a.duplicate_of:
            self.db.update(idea_id, status="await_dup")
            await self.notifier.refresh(idea_id)
            return
        self.db.update(idea_id, mode="new")
        await self.write(idea_id)

    async def write(self, idea_id: int) -> None:
        idea = self.db.get(idea_id)
        a, inc = idea.parsed, idea.incoming
        async with self.vault_lock:
            try:
                if idea.note_path:
                    self.writer.undo(idea_id)  # хвосты сырой или недописанной записи
                index = self.index_fn(self.s)
                a.links = [link for link in a.links if link.l2 in index.l2]
                append = idea.mode == "append" and (self.s.ideas_dir / f"{a.duplicate_of}.md").exists()
                if append:
                    self.writer.append(idea_id, a.duplicate_of, a, inc, index)
                else:
                    self.writer.save_new(idea_id, a, inc, index)
            except (OSError, ValueError) as e:  # ValueError: холст с битым JSON
                log.warning("write #%s: %s", idea_id, e)
                self.db.update(idea_id, status="write_failed", error=str(e))
                await self.notifier.refresh(idea_id)
                return
        self.db.update(idea_id, status="saved", mode="append" if append else "new",
                       analysis=json.dumps(a.to_dict(), ensure_ascii=False), error=None)
        await self.notifier.refresh(idea_id)

    async def resolve_duplicate(self, idea_id: int, append: bool) -> None:
        if self.db.get(idea_id).status != "await_dup":
            return
        self.db.update(idea_id, mode="append" if append else "new")
        await self.write(idea_id)

    async def relink(self, idea_id: int, l2_names: list[str]) -> None:
        idea = self.db.get(idea_id)
        if idea.status != "saved" or idea.mode != "new":
            return
        a = idea.parsed
        index = self.index_fn(self.s)
        try:
            links = await self.brain.pick_l3(a, l2_names, index)
        except BrainError as e:
            log.warning("pick_l3 #%s: %s", idea_id, e)
            links = [Link(n) for n in l2_names if n in index.l2]
        async with self.vault_lock:
            self.writer.relink(idea_id, links, index)
        a.links = links
        self.db.update(idea_id, analysis=json.dumps(a.to_dict(), ensure_ascii=False))
        await self.notifier.refresh(idea_id)

    async def deep(self, idea_id: int) -> None:
        idea = self.db.get(idea_id)
        if idea.status != "saved" or idea.deep:
            return
        try:
            md = await self.brain.deep_dive(idea.parsed)
        except BrainError as e:
            await self.notifier.error(idea_id, f"Не вышло копнуть глубже: {e}")
            await self.notifier.refresh(idea_id)
            return
        async with self.vault_lock:
            self.writer.add_deep(idea_id, md)
        await self.notifier.refresh(idea_id)

    async def undo(self, idea_id: int) -> None:
        async with self.vault_lock:
            self.writer.undo(idea_id)
        self.db.update(idea_id, status="undone")
        await self.notifier.refresh(idea_id)
```

```python
"""Тексты карточек и клавиатуры. Без длинного тире."""
from __future__ import annotations

from aiogram.types import InlineKeyboardButton as B
from aiogram.types import InlineKeyboardMarkup

from .db import Idea

ICON = {"бизнес": "💼", "контент": "🎬", "мысль": "🧠", "прочее": "🎲"}


def card_text(idea: Idea) -> str:
    a = idea.parsed
    if idea.status == "saved" and a:
        head = f"{ICON.get(a.type, '💡')} {a.type} · тема «{a.theme}»" + (f" · оценка {a.score}/10" if a.score else "")
        lines = [f"💡 {a.title}", head]
        if idea.mode == "append":
            lines.insert(0, f"➕ Дополнил «{a.duplicate_of}»")
        if a.links:
            lines.append("🔗 " + ", ".join(l.l2 + (f" → «{l.l3}»" if l.l3 else "") for l in a.links))
        else:
            lines.append("🔗 без связей с другими холстами")
        if a.unknown_tags:
            lines.append("⚠️ " + ", ".join(f"холст #{t} не нашёл" for t in a.unknown_tags))
        if idea.deep:
            lines.append("🔍 дополнено")
        return "\n".join(lines)
    return {
        "new": "⏳ Разбираю…",
        "raw": f"📝 Сохранил без разбора ({idea.error}). Лежит в «Разное» с тегом #неразобрано.",
        "stt_failed": f"🎙 Не смог расшифровать голос ({idea.error}). Файл сохранён, можно повторить.",
        "await_dup": f"🤔 Похоже на «{a.duplicate_of if a else ''}». Дополнить её или завести новую?",
        "write_failed": f"⚠️ Не смог записать в vault ({idea.error}). Повторю сам через 5 минут.",
        "undone": "↩️ Отменено, из vault убрал.",
    }.get(idea.status, idea.status)


def card_kb(idea: Idea) -> InlineKeyboardMarkup | None:
    i = idea.id
    rows: list[list[B]] = []
    if idea.status == "saved" and idea.mode == "append":
        rows = [[B(text="↩️ Отменить", callback_data=f"un:{i}")]]
    elif idea.status == "saved":
        top = [B(text="🔁 Поменять холсты", callback_data=f"ch:{i}")]
        if not idea.deep:
            top.append(B(text="🔍 Копнуть глубже", callback_data=f"dp:{i}"))
        rows = [top, [B(text="↩️ Отменить", callback_data=f"un:{i}")]]
    elif idea.status == "raw":
        rows = [[B(text="🔄 Разобрать ещё раз", callback_data=f"rt:{i}"), B(text="↩️ Отменить", callback_data=f"un:{i}")]]
    elif idea.status == "stt_failed":
        rows = [[B(text="🔄 Повторить", callback_data=f"rt:{i}")]]
    elif idea.status == "await_dup":
        rows = [[B(text="➕ Дополнить", callback_data=f"du:{i}:a"), B(text="🆕 Новая", callback_data=f"du:{i}:n")]]
    return InlineKeyboardMarkup(inline_keyboard=rows) if rows else None


def l2_kb(idea_id: int, names: list[str], selected: set[int]) -> InlineKeyboardMarkup:
    rows = [[B(text=("✅ " if n in selected else "") + name, callback_data=f"tg:{idea_id}:{n}")]
            for n, name in enumerate(names)]
    rows.append([B(text="✅ Готово", callback_data=f"ok:{idea_id}"), B(text="✖️ Назад", callback_data=f"bk:{idea_id}")])
    return InlineKeyboardMarkup(inline_keyboard=rows)
```

```python
from aiogram.exceptions import TelegramBadRequest
from aiogram.filters import CommandStart
from aiogram.types import (CallbackQuery, Message, MessageOriginChannel, MessageOriginChat,
                           MessageOriginHiddenUser, MessageOriginUser)

from .collector import Collector
from .config import Settings
from .db import DB
from .models import Part
from .pipeline import Pipeline
from .vault.index import build_index

log = logging.getLogger(__name__)
HELLO = ("Кидай идею: текст, голос или форвард (можно с комментарием, склею всё, что пришло за 5 секунд). "
         "Хочешь привязать к конкретному холсту, добавь #название, например #медиабаинг или #личный_бренд.")


def origin_name(o) -> str:
    if isinstance(o, MessageOriginChannel):
        return o.chat.title or "канал"
    if isinstance(o, MessageOriginChat):
        return o.sender_chat.title or "чат"
    if isinstance(o, MessageOriginUser):
        return o.sender_user.full_name
    if isinstance(o, MessageOriginHiddenUser):
        return o.sender_user_name
    return "неизвестный источник"


async def to_part(m: Message, s: Settings) -> Part | None:
    origin = origin_name(m.forward_origin) if m.forward_origin else None
    text = m.text or m.caption or ""
    folder = s.media_dir / f"{m.chat.id}_{m.message_id}"
    media, name = None, ""
    if m.voice:
        media, name = m.voice, "voice.ogg"
    elif m.audio:
        media, name = m.audio, "audio" + (Path(m.audio.file_name).suffix if m.audio.file_name else ".mp3")
    elif m.video_note:
        media, name = m.video_note, "note.mp4"
    if media:
        folder.mkdir(parents=True, exist_ok=True)
        await m.bot.download(media, destination=folder / name)
        return Part(text=text, origin=origin, voice=str(folder / name))
    if m.photo:
        folder.mkdir(parents=True, exist_ok=True)
        await m.bot.download(m.photo[-1], destination=folder / "photo.jpg")
        return Part(text=text, origin=origin, photo=str(folder / "photo.jpg"))
    if m.video or m.animation:
        return Part(text=text, origin=origin, has_video=True)
    if text:
        return Part(text=text, origin=origin)
    return None


class TgNotifier:
    def __init__(self, bot: Bot, db: DB):
        self.bot, self.db = bot, db

    async def _show(self, idea_id: int, text: str, kb) -> None:
        idea = self.db.get(idea_id)
        if idea.card_msg_id:
            try:
                await self.bot.edit_message_text(text=text, chat_id=idea.chat_id, message_id=idea.card_msg_id,
                                                 reply_markup=kb)
                return
            except TelegramBadRequest as e:
                if "not modified" in str(e):
                    return
                log.warning("карточку #%s не отредактировать: %s", idea_id, e)
        msg = await self.bot.send_message(idea.chat_id, text, reply_markup=kb)
        self.db.update(idea_id, card_msg_id=msg.message_id)

    async def refresh(self, idea_id: int) -> None:
        idea = self.db.get(idea_id)
        await self._show(idea_id, card_text(idea), card_kb(idea))

    async def busy(self, idea_id: int, text: str) -> None:
        await self._show(idea_id, text, None)

    async def error(self, idea_id: int, text: str) -> None:
        idea = self.db.get(idea_id)
        await self.bot.send_message(idea.chat_id, text)


def build_router(s: Settings, db: DB, pipeline: Pipeline, collector: Collector) -> Router:
    r = Router()
    r.message.filter(F.from_user.id == s.owner_id)
    r.callback_query.filter(F.from_user.id == s.owner_id)
    picks: dict[int, tuple[list[str], set[int]]] = {}
    notifier = pipeline.notifier

    @r.message(CommandStart())
    async def start(m: Message):
        await m.answer(HELLO)

    @r.message()
    async def incoming(m: Message):
        part = await to_part(m, s)
        if part is None:
            await m.answer("Такое не понимаю: нужен текст, голос, фото или форвард.")
            return
        collector.add(m.chat.id, part)

    def _id(c: CallbackQuery) -> int:
        return int(c.data.split(":")[1])

    @r.callback_query(F.data.regexp(r"^ch:\d+$"))
    async def change(c: CallbackQuery):
        i = _id(c)
        idea = db.get(i)
        if idea.status != "saved" or idea.mode != "new":
            await c.answer("Для этой идеи холсты не меняются")
            return
        names = sorted(build_index(s).l2)
        current = {l.l2 for l in idea.parsed.links}
        picks[i] = (names, {n for n, name in enumerate(names) if name in current})
        await c.message.edit_reply_markup(reply_markup=l2_kb(i, *picks[i]))
        await c.answer()

    @r.callback_query(F.data.regexp(r"^tg:\d+:\d+$"))
    async def toggle(c: CallbackQuery):
        i, n = _id(c), int(c.data.split(":")[2])
        if i not in picks:
            await c.answer("Список устарел, нажми «Поменять холсты» ещё раз")
            return
        picks[i][1].symmetric_difference_update({n})
        await c.message.edit_reply_markup(reply_markup=l2_kb(i, *picks[i]))
        await c.answer()

    @r.callback_query(F.data.regexp(r"^ok:\d+$"))
    async def done(c: CallbackQuery):
        i = _id(c)
        names, sel = picks.pop(i, ([], set()))
        await c.answer("Перевешиваю…")
        await notifier.busy(i, "🔁 Перевешиваю связи…")
        pipeline.spawn(pipeline.relink(i, [names[n] for n in sorted(sel)]))

    @r.callback_query(F.data.regexp(r"^bk:\d+$"))
    async def back(c: CallbackQuery):
        picks.pop(_id(c), None)
        await notifier.refresh(_id(c))
        await c.answer()

    @r.callback_query(F.data.regexp(r"^dp:\d+$"))
    async def deep(c: CallbackQuery):
        await c.answer("Копаю, это 1-3 минуты")
        await notifier.busy(_id(c), "🔍 Копаю глубже, 1-3 минуты…")
        pipeline.spawn(pipeline.deep(_id(c)))

    @r.callback_query(F.data.regexp(r"^un:\d+$"))
    async def undo(c: CallbackQuery):
        await c.answer()
        pipeline.spawn(pipeline.undo(_id(c)))

    @r.callback_query(F.data.regexp(r"^du:\d+:[an]$"))
    async def dup(c: CallbackQuery):
        await c.answer()
        await notifier.busy(_id(c), "⏳ Записываю…")
        pipeline.spawn(pipeline.resolve_duplicate(_id(c), append=c.data.endswith(":a")))

    @r.callback_query(F.data.regexp(r"^rt:\d+$"))
    async def retry(c: CallbackQuery):
        await c.answer()
        await notifier.busy(_id(c), "⏳ Разбираю…")
        pipeline.submit(_id(c))

    return r
```

```python
"""Один процесс: поллинг бота + очередь + повторы."""
from __future__ import annotations

import asyncio
import logging
from logging.handlers import RotatingFileHandler

from aiogram import Bot, Dispatcher

from .bot import TgNotifier, build_router
from .brain import Brain
from .collector import Collector
from .config import load_settings
from .db import DB
from .models import Incoming
from .pipeline import Pipeline
from .stt import transcribe
from .vault.sync import Writer


async def main() -> None:
    s = load_settings()
    s.logs_dir.mkdir(exist_ok=True)
    logging.basicConfig(
        level=logging.INFO, format="%(asctime)s %(levelname)s %(name)s: %(message)s",
        handlers=[RotatingFileHandler(s.logs_dir / "ideabot.log", maxBytes=1_000_000, backupCount=3, encoding="utf-8"),
                  logging.StreamHandler()])
    logging.getLogger("aiogram.event").setLevel(logging.WARNING)
    if not s.bot_token or not s.owner_id:
        raise SystemExit("Заполни BOT_TOKEN и OWNER_ID в .env")
    if not s.ideas_canvas_path.exists():
        raise SystemExit(f"Не нашёл холст {s.ideas_canvas_path}")
    db = DB(s.db_path)
    bot = Bot(s.bot_token)
    pipeline = Pipeline(s, db, Writer(s, db), Brain(s), lambda p: transcribe(s, p), TgNotifier(bot, db))

    async def on_flush(chat_id, parts):
        idea_id = db.create(chat_id, Incoming(parts))
        msg = await bot.send_message(chat_id, "⏳ Разбираю…")
        db.update(idea_id, card_msg_id=msg.message_id)
        pipeline.submit(idea_id)

    dp = Dispatcher()
    dp.include_router(build_router(s, db, pipeline, Collector(s.collect_window, on_flush)))
    pipeline.startup()
    logging.getLogger("ideabot").info("Idea Bot запущен, vault: %s", s.vault)
    try:
        await asyncio.gather(pipeline.run(), pipeline.retry_loop(), dp.start_polling(bot, handle_signals=False))
    finally:
        await bot.session.close()


if __name__ == "__main__":
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        pass
```
